Зверніть увагу на те, що відбувається, коли круг повністю заходить всередину багатокутника (ми це невдовзі розберемо).

Багатокутник і Круг

Завдання з перевірки зіткнення круга з багатокутником дуже легко зібрати з уже готових деталей! Оскільки контур багатокутника складається з відрізків (ребер), розвʼязок можна звести до серії перевірок зіткнення лінії з кругом — по одній для кожного ребра багатокутника. Для цього нам достатньо в циклі перебрати всі сусідні пари вершин currVertex і nextVertex (як ми робили в розділі «Багатокутник і Точка») та для кожного ребра викликати функцію collideLineCircle(): 

const isCollision = collideLineCircle(
  currVertex.x, currVertex.y,
  nextVertex.x, nextVertex.y,
  cx, cy, radius
);

if (isCollision) {
  return true;
}

Щойно хоча б одне ребро багатокутника перетнеться з кругом, функція негайно поверне true, навіть не витрачаючи час на перевірку решти сторін. У цьому й полягає краса модульного підходу: складні алгоритми будуються з простих, уже перевірених цеглинок!

Повний приклад коду:

// координати та радіус рухомого круга
let cx = 0;
let cy = 0;
const radius = 30;

// масив векторів вершин багатокутника
const vertices = [];

function setup() {
  createCanvas(window.innerWidth, window.innerHeight);
  noStroke();

  // задаємо вершини багатокутника (тут це трапеція)
  vertices[0] = createVector(width / 2 - 100, height / 2 - 100);
  vertices[1] = createVector(width / 2 + 100, height / 2 - 100);
  vertices[2] = createVector(width / 2 + 50,  height / 2 + 100);
  vertices[3] = createVector(width / 2 - 50,  height / 2 + 100);
}

function draw() {
  background(255);

  // оновлюємо координати центра круга положенням курсора
  cx = mouseX;
  cy = mouseY;

  // перевіряємо зіткнення багатокутника та круга
  const isColliding = collidePolyCircle(vertices, cx, cy, radius);

  // змінюємо колір багатокутника при зіткненні
  if (isColliding) {
    fill(255, 150, 0);
  } else {
    fill(0, 150, 255);
  }

  // малюємо багатокутник
  noStroke();
  beginShape();
  for (const v of vertices) {
    vertex(v.x, v.y);
  }
  endShape(CLOSE);

  // малюємо круг
  fill(0, 150);
  ellipse(cx, cy, radius * 2, radius * 2);
}

// перевірка на зіткнення між багатокутником і кругом
function collidePolyCircle(vertices, cx, cy, radius) {
  // перебираємо всі ребра багатокутника
  let next = 0;
  for (let current = 0; current < vertices.length; current++) {
    next = current + 1;

    if (next === vertices.length) {
      next = 0;
    }

    const currVertex = vertices[current];
    const nextVertex = vertices[next];

    // перевіряємо перетин круга з поточним ребром
    const isCollision = collideLineCircle(
      currVertex.x, currVertex.y,
      nextVertex.x, nextVertex.y,
      cx, cy, radius
    );

    if (isCollision) {
      return true;
    }
  }

  // необов’язково: щоб врахувати випадок, коли круг повністю всередині
  // багатокутника і не торкається його сторін, розкоментуйте ці два рядки:
  // const isCenterInside = collidePolyPoint(vertices, cx, cy);
  // if (isCenterInside) return true;

  return false;
}

// перевірка на зіткнення між лінією і кругом
function collideLineCircle(x1, y1, x2, y2, cx, cy, radius) {
  // якщо один з кінців всередині круга,
  // тоді одразу повертаємо true
  const inside1 = collidePointCircle(x1, y1, cx, cy, radius);
  const inside2 = collidePointCircle(x2, y2, cx, cy, radius);
  
  if (inside1 || inside2) {
    return true;
  }

  // отримання довжини лінії
  let distX = x1 - x2;
  let distY = y1 - y2;
  const lineLen = sqrt((distX * distX) + (distY * distY));

  // скалярний добуток прямої та круга
  const dot = (((cx - x1) * (x2 - x1)) + ((cy - y1) * (y2 - y1))) / pow(lineLen, 2);

  // знаходження найближчої точки на прямій
  const closestX = x1 + (dot * (x2 - x1));
  const closestY = y1 + (dot * (y2 - y1));

  // ця точка дійсно знаходиться на відрізку?
  // якщо так, продовжуємо, але повертаємо false  const isOnSegment = collideLinePoint(x1, y1, x2, y2, closestX, closestY);
  if (!isOnSegment) {
    return false;
  }

  // необов’язково: малюємо маркер найближчої точки на ребрі
  fill(255, 0, 0);
  noStroke();
  ellipse(closestX, closestY, 20, 20);

  // обчислення відстані до найближчої точки
  distX = closestX - cx;
  distY = closestY - cy;
  const distance = sqrt((distX * distX) + (distY * distY));

  // чи круг має перетин з лінією?
  if (distance <= radius) {
    return true;
  }

  return false;
}

// перевірка на зіткнення між лінією і точкою
function collideLinePoint(x1, y1, x2, y2, px, py) {
  // обчислення відстані від точки до двох кінців відрізка
  const d1 = dist(px, py, x1, y1);
  const d2 = dist(px, py, x2, y2);

  // обчислення довжини відрізка
  const lineLen = dist(x1, y1, x2, y2);

  // невеликий додатковий буфер для збільшення області перетину
  const buffer = 0.1;

  // якщо сума двох відстаней дорівнює довжині відрізка, тоді точка знаходиться на відрізку!
  // зауважте, що тут ми додатково використовуємо буфер для збільшення радіуса зіткнення
  if (d1 + d2 >= lineLen - buffer && d1 + d2 <= lineLen + buffer) {
    return true;
  }

  return false;
}

// перевірка на зіткнення між точкою і кругом
function collidePointCircle(px, py, cx, cy, radius) {
  const distX = px - cx;
  const distY = py - cy;
  const distance = sqrt((distX * distX) + (distY * distY));

  // якщо відстань менша за радіус кола, значить точка всередині!
  if (distance <= radius) {
    return true;
  }

  return false;
}

// перевірка на зіткнення між багатокутником та точкою (опціонально)
// потрібно, лише якщо ви збираєтеся перевірити, чи круг знаходиться ВСЕРЕДИНІ багатокутника
function collidePolyPoint(vertices, px, py) {
  let isCollision = false;

  // перебір кожної вершини з використанням наступної вершини в списку
  let next = 0;
  for (let current = 0; current < vertices.length; current++) {
    // отримання наступної вершини зі списку
    next = current + 1;

    // коли дійшли до останньої вершини, беремо першу під індексом 0
    if (next === vertices.length) {
      next = 0;
    }

    const currVertex = vertices[current];
    const nextVertex = vertices[next];

    if (
      ((currVertex.y >= py && nextVertex.y < py) || (currVertex.y < py && nextVertex.y >= py)) &&
      (px < (nextVertex.x - currVertex.x) * (py - currVertex.y) / (nextVertex.y - currVertex.y) + currVertex.x)
    ) {
      isCollision = !isCollision;
    }
  }

  return isCollision;
}

Граничний випадок: круг повністю всередині багатокутника

Спробуйте в інтерактивному прикладі вгорі завести круг глибоко всередину трапеції так, щоб він не торкався жодної з її чотирьох сторін. Помітили, що підсвічування зникло? Такі нестандартні ситуації в програмуванні називають граничними випадками (edge cases).

Чому так сталося? Бо наш цикл перевіряє лише перетин круга зі стінками багатокутника. У багатьох іграх цього цілком достатньо: наприклад, якщо астероїд (круг) летить у космічний корабель (багатокутник), він неодмінно спершу вдариться об зовнішню обшивку корабля, і вибух спрацює ще на межі.

Але якщо вам потрібно, щоб зіткнення фіксувалося навіть тоді, коли круг повністю сховався всередині багатокутника, достатньо після циклу for перевірити, чи лежить центр круга (cx, cy) всередині багатокутника за допомогою функції collidePolyPoint() із попереднього розділу: 

const isCenterInside = collidePolyPoint(vertices, cx, cy);

if (isCenterInside) {
  return true;
}

Зверніть увагу, що ми розміщуємо цю додаткову перевірку після циклу перебору ребер: завдяки цьому вона виконується лише один раз і тільки тоді, коли круг не зачепив жодної сторони.
