Багатокутник і Лінія

Перевірка перетину лінії з багатокутником працює за тим самим принципом, що й у розділі «Лінія і Прямокутник»: ми по черзі обходимо кожну сторону багатокутника і викликаємо для неї функцію перетину двох відрізків collideLineLine().

Щоб урізноманітнити наші сцени, замість трапеції згенеруймо правильний шістнадцятикутник (гексадекагон) за допомогою тригонометричних функцій cos() та sin(). Повний оберт кола становить TWO_PI радіанів (360°), тож якщо поділити його на 16 рівних кроків, ми отримаємо вершини, рівномірно розташовані по колу радіусом 100 пікселів: 

const verticesCount = 16;
const vertices = [];

const angleStep = TWO_PI / verticesCount;
// створюємо вершини правильного багатокутника
for (let i = 0; i < verticesCount; i++) {
  const angle = angleStep * i;
  const x = width / 2 + cos(angle) * 100;
  const y = height / 2 + sin(angle) * 100;
  vertices.push(createVector(x, y));
}

Далі в циклі for перебираємо всі ребра багатокутника, отримуючи поточну та наступну вершини: 

let next = 0;
for (let current = 0; current < vertices.length; current++) {
  next = current + 1;
  if (next === vertices.length) {
    next = 0;
  }

  const currVertex = vertices[current];
  const nextVertex = vertices[next];
}

Залишається передати координати нашого відрізка (x1, y1, x2, y2) та координати поточного ребра (x3, y3, x4, y4) у функцію collideLineLine(): 

const x3 = currVertex.x;
const y3 = currVertex.y;
const x4 = nextVertex.x;
const y4 = nextVertex.y;

const isHit = collideLineLine(x1, y1, x2, y2, x3, y3, x4, y4);
if (isHit) {
  return true;
}

Якщо знайдено перетин хоча б з одним ребром, ми одразу повертаємо true і заощаджуємо час на перевірці решти сторін. Якщо ж цикл завершився без жодного перетину — повертаємо false.

Повний код прикладу:

// координати відрізка з керованим кінцем
let x1 = 0;
let y1 = 0;
const x2 = 20;
const y2 = 20;

const verticesCount = 16; // кількість вершин багатокутника
const vertices = [];      // масив векторів вершин

function setup() {
  createCanvas(window.innerWidth, window.innerHeight);
  strokeWeight(15); // збільшена товщина для кращої видимості лінії
  noCursor();

  // створюємо вершини правильного 16-кутника
  const angleStep = TWO_PI / verticesCount;
  for (let i = 0; i < verticesCount; i++) {
    const angle = angleStep * i;
    const x = width / 2 + cos(angle) * 100;
    const y = height / 2 + sin(angle) * 100;
    vertices.push(createVector(x, y));
  }
}

function draw() {
  background(255);

  // оновлюємо один кінець лінії координатами курсора
  x1 = mouseX;
  y1 = mouseY;

  // перевіряємо перетин багатокутника з лінією
  const isColliding = collidePolyLine(vertices, x1, y1, x2, y2);

  // змінюємо колір багатокутника при зіткненні
  if (isColliding) {
    fill(255, 150, 0);
  } else {
    fill(0, 150, 255);
  }

  // малюємо багатокутник
  noStroke();
  beginShape();
  for (const v of vertices) {
    vertex(v.x, v.y);
  }
  endShape(CLOSE);

  // малюємо лінію
  stroke(0, 150);
  line(x1, y1, x2, y2);
}

// перевірка на зіткнення між багатокутником та лінією
function collidePolyLine(vertices, x1, y1, x2, y2) {
  // перебираємо всі ребра багатокутника
  let next = 0;
  for (let current = 0; current < vertices.length; current++) {
    next = current + 1;
    if (next === vertices.length) {
      next = 0;
    }

    const currVertex = vertices[current];
    const nextVertex = vertices[next];

    // координати поточного ребра багатокутника
    const x3 = currVertex.x;
    const y3 = currVertex.y;
    const x4 = nextVertex.x;
    const y4 = nextVertex.y;

    // перевіряємо перетин лінії з поточним ребром
    const isHit = collideLineLine(x1, y1, x2, y2, x3, y3, x4, y4);
    if (isHit) {
      return true;
    }
  }

  return false;
}

// перевірка на перетин двох відрізків
function collideLineLine(x1, y1, x2, y2, x3, y3, x4, y4) {
  const uA = ((x4 - x3) * (y1 - y3) - (y4 - y3) * (x1 - x3)) / ((y4 - y3) * (x2 - x1) - (x4 - x3) * (y2 - y1));
  const uB = ((x2 - x1) * (y1 - y3) - (y2 - y1) * (x1 - x3)) / ((y4 - y3) * (x2 - x1) - (x4 - x3) * (y2 - y1));

  if (uA >= 0 && uA <= 1 && uB >= 0 && uB <= 1) {
    return true;
  }

  return false;
}
