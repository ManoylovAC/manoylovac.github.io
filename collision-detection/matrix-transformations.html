Робота з матричними перетвореннями

Досі ми задавали всі фігури безпосередньо їхніми екранними координатами x/y. Проте у творчому програмуванні на p5.js об’єкти дуже часто переміщують і обертають за допомогою матричних перетворень — функцій push() / pop(), translate() та rotate().

У чому тут підступ для наших алгоритмів зіткнення? Коли ви викликаєте translate() та rotate(), ви повертаєте саму координатну сітку полотна, а числа у змінних вашої фігури залишаються незмінними! Тобто на екрані квадрат уже повернувся на 45° і змістився в центр, але його локальні вершини все ще думають, що вони лежать у точках (-50, -50).

Підказка: якщо ви хочете освіжити в пам’яті, як працюють трансформації координатної сітки в p5.js, перегляньте чудовий відеоурок Деніела Шиффмана Translate, Rotate, Push/Pop на сайті The Coding Train.

Переведення локальних координат в екранні

На щастя, ми можемо скористатися невеликим розширенням для p5.js — функцією screenPosition(). Ви передаєте їй локальні координати (x, y) точки всередині блоку push() / pop(), а вона множить їх на поточну матрицю трансформацій і повертає фактичні екранні координати цієї точки відносно лівого верхнього кута полотна!

Оскільки після обертання звичайний прямокутник перестає бути вирівняним за осями координат (його сторони стають похилими), найзручніше розглядати будь-яку повернуту фігуру як багатокутник і перевіряти її універсальною функцією collidePolyPoly()!

Спочатку задамо вершини квадрата розміром 100×100 пікселів у його локальній системі координат навколо власного центра (0, 0): 

const squarePoints1 = [
  createVector(-50, -50),
  createVector( 50, -50),
  createVector( 50,  50),
  createVector(-50,  50),
];

Напишимо допоміжну функцію pointsToScreenCoords(), яка в поточному кадрі пропускає кожну локальну вершину фігури через screenPosition() і повертає масив її реальних екранних координат: 

// перетворює масив локальних вершин на фактичні координати полотна
// з урахуванням активних translate() та rotate()
function pointsToScreenCoords(points) {
  const screenPoints = new Array(points.length);

  for (let i = 0; i < points.length; i++) {
    const screenPoint = screenPosition(points[i].x, points[i].y);
    screenPoints[i] = createVector(screenPoint.x, screenPoint.y);
  }

  return screenPoints;
}

Отримавши екранні координати обох повернутих квадратів (square1Screen та square2Screen), ми просто передаємо їх у функцію collidePolyPoly() — і маємо ідеально точне виявлення зіткнень під будь-яким кутом обертання!

Нижче наведено повний код програми (допоміжна функція addScreenPositionFunction() підключена окремим скриптом): 

// масиви локальних вершин двох квадратів
let squarePoints1;
let squarePoints2;

let angle = 0;

function setup() {
  createCanvas(window.innerWidth, window.innerHeight);
  noCursor();

  // активуємо функцію screenPosition() із підключеного додаткового коду
  // джерело: https://github.com/bohnacker/p5js-screenPosition
  addScreenPositionFunction();

  squarePoints1 = [
    createVector(-50, -50),
    createVector( 50, -50),
    createVector( 50,  50),
    createVector(-50,  50),
  ];
  squarePoints2 = [
    createVector(-100, -100),
    createVector( 100, -100),
    createVector( 100,  100),
    createVector(-100,  100),
  ];
}

function draw() {
  background(255);

  // збільшуємо кут обертання в кожному кадрі
  angle += 0.02;

  // переміщуємо та повертаємо систему координат для першого квадрата
  push();
  translate(width / 2 - 100, height / 2);
  rotate(angle);

  // зчитуємо фактичні екранні координати першого квадрата після трансформацій
  const square1Screen = pointsToScreenCoords(squarePoints1);

  // малюємо перший квадрат
  fill(0, 150);
  noStroke();
  beginShape();
  for (const pt of squarePoints1) {
    vertex(pt.x, pt.y);
  }
  endShape(CLOSE);
  pop();

  // переміщуємо та повертаємо систему координат для другого квадрата
  push();
  translate(width / 2 + 100, height / 2);
  rotate(angle);

  // зчитуємо фактичні екранні координати другого квадрата
  const square2Screen = pointsToScreenCoords(squarePoints2);

  // перевіряємо зіткнення двох повернутих квадратів як багатокутників
  const isColliding = collidePolyPoly(square1Screen, square2Screen);

  // змінюємо колір другого квадрата при зіткненні
  if (isColliding) {
    fill(255, 150, 0);
  } else {
    fill(0, 150, 255);
  }

  // малюємо другий квадрат
  beginShape();
  for (const pt of squarePoints2) {
    vertex(pt.x, pt.y);
  }
  endShape(CLOSE);
  pop();
}

// перетворення локальних вершин на екранні координати полотна
function pointsToScreenCoords(points) {
  const screenPoints = new Array(points.length);

  for (let i = 0; i < points.length; i++) {
    const screenPoint = screenPosition(points[i].x, points[i].y);
    screenPoints[i] = createVector(screenPoint.x, screenPoint.y);
  }

  return screenPoints;
}

// перевірка на зіткнення між двома багатокутниками
function collidePolyPoly(p1, p2) {
  let next = 0;
  for (let current = 0; current < p1.length; current++) {
    next = current + 1;
    if (next === p1.length) {
      next = 0;
    }

    const currentVertex = p1[current];
    const nextVertex = p1[next];

    const isCollision = collidePolyLine(
      p2,
      currentVertex.x, currentVertex.y,
      nextVertex.x, nextVertex.y
    );
    if (isCollision) {
      return true;
    }
  }

  // необов’язково: перевіряємо, чи один багатокутник всередині іншого
  const isInside = collidePolyPoint(p1, p2[0].x, p2[0].y);
  if (isInside) {
    return true;
  }

  return false;
}

// перевірка на зіткнення між багатокутником та лінією
function collidePolyLine(vertices, x1, y1, x2, y2) {
  let next = 0;
  for (let current = 0; current < vertices.length; current++) {
    next = current + 1;
    if (next === vertices.length) {
      next = 0;
    }

    const currentVertex = vertices[current];
    const nextVertex = vertices[next];

    const x3 = currentVertex.x;
    const y3 = currentVertex.y;
    const x4 = nextVertex.x;
    const y4 = nextVertex.y;

    const isHit = collideLineLine(x1, y1, x2, y2, x3, y3, x4, y4);
    if (isHit) {
      return true;
    }
  }

  return false;
}

// перевірка на перетин двох відрізків
function collideLineLine(x1, y1, x2, y2, x3, y3, x4, y4) {
  const uA = ((x4 - x3) * (y1 - y3) - (y4 - y3) * (x1 - x3)) / ((y4 - y3) * (x2 - x1) - (x4 - x3) * (y2 - y1));
  const uB = ((x2 - x1) * (y1 - y3) - (y2 - y1) * (x1 - x3)) / ((y4 - y3) * (x2 - x1) - (x4 - x3) * (y2 - y1));

  return uA >= 0 && uA <= 1 && uB >= 0 && uB <= 1;
}

// перевірка на зіткнення між багатокутником та точкою
function collidePolyPoint(vertices, px, py) {
  let isCollision = false;

  let next = 0;
  for (let current = 0; current < vertices.length; current++) {
    next = current + 1;
    if (next === vertices.length) {
      next = 0;
    }

    const currentVertex = vertices[current];
    const nextVertex = vertices[next];

    if (
      ((currentVertex.y >= py && nextVertex.y < py) || (currentVertex.y < py && nextVertex.y >= py)) &&
      (px < (nextVertex.x - currentVertex.x) * (py - currentVertex.y) / (nextVertex.y - currentVertex.y) + currentVertex.x)
    ) {
      isCollision = !isCollision;
    }
  }

  return isCollision;
}
