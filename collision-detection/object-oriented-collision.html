Перехід до об’єктно-орієнтованого зіткнення

Вітаємо! Ви пройшли через усі основні алгоритми виявлення зіткнень. Досі в кожному прикладі ми мали лише дві фігури, координати яких зберігалися в окремих глобальних змінних (rx, ry, rw, rh). Це чудово підходить для навчання, але що, як у вашій грі буде десяток платформ, сотня монет і зграя ворогів?

Створювати окремі змінні rect1x, rect2x, ..., rect100x було б справжнім кошмаром і це швидко призведе до безладу! Саме тут на допомогу приходить об’єктно-орієнтоване програмування (ООП) та класи. Якщо ви хочете глибше розібратися з класами та об’єктами в p5.js, радимо зазирнути до книги Деніела Шиффмана «Природа коду».

Розгляньте сцену вгорі: у нас є один рухомий круг і цілий масив прямокутників. Ми можемл описати кожну фігуру власним класом — Circle та Rectangle. Кожен об’єкт сам пам’ятатиме свої координати, розміри та вмітиме малювати себе на полотні.

Почнімо зі створення простого класу Circle: 

class Circle {
  constructor(x, y, radius) {
    this.x = x;           // x-координата центра
    this.y = y;           // y-координата центра
    this.radius = radius; // радіус круга
  }

  // оновлення координат круга
  update(x, y) {
    this.x = x;
    this.y = y;
  }

  // малювання круга на полотні
  display() {
    fill(0, 150);
    noStroke();
    ellipse(this.x, this.y, this.radius * 2, this.radius * 2);
  }
}

Тепер створимо клас Rectangle: 

class Rectangle {
  constructor(x, y, w, h) {
    this.x = x;          // x-координата лівого верхнього кута
    this.y = y;          // y-координата лівого верхнього кута
    this.w = w;          // ширина
    this.h = h;          // висота
    this.isHit = false;  // прапорець стану зіткнення
  }

  // метод для перевірки зіткнення цього прямокутника з об’єктом круга
  checkCollisionWithCircle(circleObj) {
    this.isHit = collideCircleRect(
      circleObj.x, circleObj.y, circleObj.radius,
      this.x, this.y, this.w, this.h
    );
  }

  // малювання прямокутника
  display() {
    // при зіткненні змінюємо колір
    if (this.isHit) {
      fill(255, 150, 0);
    } else {
      fill(0, 150, 255);
    }
    noStroke();
    rect(this.x, this.y, this.w, this.h);
  }
}

Зверніть увагу на властивість this.isHit усередині класу Rectangle (типово вона дорівнює false). Кожен прямокутник має власний індивідуальний прапорець isHit і змінює свій колір у методі display() лише тоді, коли круг торкається саме його!

А як відбувається сама перевірка? Ми додали до класу Rectangle метод checkCollisionWithCircle(circleObj), який приймає об’єкт круга і передає його властивості та власні координати прямокутника до вже знайомої нам функції collideCircleRect(): 

checkCollisionWithCircle(circleObj) {
  this.isHit = collideCircleRect(
    circleObj.x, circleObj.y, circleObj.radius,
    this.x, this.y, this.w, this.h
  );
}

Завдяки класам головна функція draw() стає неймовірно чистою та зрозумілою — незалежно від того, чи на екрані 8 прямокутників, чи 800: 

function draw() {
  background(255);

  // перебираємо всі прямокутники в масиві
  for (const rectangle of rects) {
    rectangle.checkCollisionWithCircle(circle); // перевіряємо зіткнення
    rectangle.display();                        // малюємо прямокутник
  }

  // оновлюємо та малюємо круг курсора
  circle.update(mouseX, mouseY);
  circle.display();
}

Повний код прикладу:

// змінна для об’єкта круга, яким керує курсор
let circle;
// масив для об’єктів прямокутників
const rects = new Array(8);

function setup() {
  createCanvas(window.innerWidth, window.innerHeight);

  // створюємо круг радіусом 30 пікселів
  circle = new Circle(0, 0, 30);

  // генеруємо 8 прямокутників у випадкових клітинках сітки з кроком у 50 пікселів
  for (let i = 0; i < rects.length; i++) {
    const x = round(random(50, width - 50) / 50) * 50;
    const y = round(random(50, height - 50) / 50) * 50;
    rects[i] = new Rectangle(x, y, 50, 50);
  }
}

function draw() {
  background(255);

  // перебираємо всі прямокутники
  for (const rectangle of rects) {
    rectangle.checkCollisionWithCircle(circle); // перевірка на зіткнення
    rectangle.display();                        // відображення прямокутника
  }

  // оновлюємо положення круга координатами курсора і малюємо його
  circle.update(mouseX, mouseY);
  circle.display();
}

class Circle {
  constructor(x, y, radius) {
    this.x = x;           // x-координата центра
    this.y = y;           // y-координата центра
    this.radius = radius; // радіус
  }

  update(x, y) {
    this.x = x;
    this.y = y;
  }

  display() {
    fill(0, 150);
    noStroke();
    ellipse(this.x, this.y, this.radius * 2, this.radius * 2);
  }
}

class Rectangle {
  constructor(x, y, w, h) {
    this.x = x;         // x-координата лівого верхнього кута
    this.y = y;         // y-координата лівого верхнього кута
    this.w = w;         // ширина
    this.h = h;         // висота
    this.isHit = false; // чи є зіткнення
  }

  // перевірка на зіткнення з об’єктом круга
  checkCollisionWithCircle(circleObj) {
    this.isHit = collideCircleRect(
      circleObj.x, circleObj.y, circleObj.radius,
      this.x, this.y, this.w, this.h
    );
  }

  display() {
    if (this.isHit) {
      fill(255, 150, 0);
    } else {
      fill(0, 150, 255);
    }
    noStroke();
    rect(this.x, this.y, this.w, this.h);
  }
}

// перевірка на зіткнення між кругом і прямокутником
function collideCircleRect(cx, cy, radius, rx, ry, rw, rh) {
  let testX = cx;
  let testY = cy;

  if (cx < rx) {
    testX = rx;          // лівий край
  } else if (cx > rx + rw) {
    testX = rx + rw;     // правий край
  }

  if (cy < ry) {
    testY = ry;          // верхній край
  } else if (cy > ry + rh) {
    testY = ry + rh;     // нижній край
  }

  const distX = cx - testX;
  const distY = cy - testY;
  const distance = sqrt((distX * distX) + (distY * distY));

  return distance <= radius;
}

У реальних проєктах класи та допоміжні функції з виявлення зіткнень зручно виносити в окремі js-файли (наприклад, зібрати всі функції перевірки в один файл collision-detection-functions.js), щоб основний файл програми залишався компактним.
